pliki cookies to niewielkie pliki tekstowe
część specyfikacji protokołu HTTP
służą głownie do indentyfikacji użytkownika

pliki zawierą:
nazwe serwera
date wygaśnięcia pliku

setcookie(nazwa[wartość, czas_trwania, ścieżka_dostępu, domena, bezpieczeństwo, tylko_http])

bezpieczeństwo true(tylko HTTPS)/false pozostałe

$_COOKIE["nazwa pliku"]

<?php

setcookie("pismo","na skróty",time()+3600);
if(isset($_COOKIE["pismo"])) {
echo "jesteś naszym stałym klietem";
} else {
    echo "Witamy po raz pierwszy na stronie";
}

?>
